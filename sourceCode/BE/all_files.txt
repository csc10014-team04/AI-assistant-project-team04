class NluResult(BaseModel):
    intent: Intent = "unknown"
    title: str | None = Field(default=None, max_length=255)
    duration_minutes: int | None = Field(default=None, ge=15, le=720)
    priority: int | None = Field(default=None, ge=1, le=3)
    preferred_time_of_day: TimeOfDay | None = None
    avoid_time_of_day: TimeOfDay | None = None
    date_hint: str | None = Field(default=None, max_length=100)
    window_start: date | None = None  # resolved deterministically from date_hint (P2-3)
    window_end: date | None = None
    date_assumed: bool = False  # True when a default window was applied
    needs_clarification: bool = False
    clarification_question: str | None = None
    profanity_found: bool = False
    fallback_mode: bool = False  # True when the llm failed and rules produced this result
    source: Literal["llm", "rules"] = "llm"

_DURATION_RE = re.compile(
    r"(\d+(?:[.,]\d+)?)\s*(tiếng|tieng|giờ|gio|h|hours?|hrs?|phút|phut|p|mins?|minutes?)\b",
    re.IGNORECASE,
)
_DATE_HINT_RE = re.compile(
    r"(\d{1,2}/\d{1,2}(?:/\d{2,4})?|hôm nay|hom nay|ngày mai|ngay mai|ngày kia|tuần này|tuan nay|tuần sau|tuan sau|"
    r"cuối tuần|cuoi tuan|cuối tháng|tháng sau|"
    r"thứ\s*(?:hai|ba|tư|năm|sáu|bảy|[2-7])|chủ nhật|today|tomorrow|this week|next week)",
    re.IGNORECASE,
)
_TIME_OF_DAY = {
    "morning": ("sáng", "sang", "morning"),
    "afternoon": ("chiều", "chieu", "afternoon"),
    "evening": ("tối", "toi", "evening", "night"),
}
_NEGATION = ("không", "khong", "ko", "đừng", "dung", "tránh", "tranh", "not", "avoid", "no")
_HIGH_PRIORITY = ("gấp", "gap", "quan trọng", "quan trong", "urgent", "asap", "deadline", "thi", "exam")


def extract_duration_minutes(text: str) -> int | None:
    """
    input: text (str) sanitized prompt.
    output: int minutes (15..720) from the first "<number> <unit>" phrase, else None.
    typical flow: "1,5 tiếng" -> 90; "45 phút" -> 45; "3h" -> 180.
    """
    match = _DURATION_RE.search(text)
    if not match:
        return None
    value = float(match.group(1).replace(",", "."))
    unit = match.group(2).lower()
    minutes = round(value * (1 if unit in {"phút", "phut", "p", "min", "mins", "minute", "minutes"} else 60))
    return minutes if 15 <= minutes <= 720 else None


def _time_of_day_flags(text: str) -> tuple[TimeOfDay | None, TimeOfDay | None]:
    """returns (preferred, avoided) by checking a negation word right before the period word."""
    lowered = text.lower()
    preferred: TimeOfDay | None = None
    avoided: TimeOfDay | None = None
    for period, words in _TIME_OF_DAY.items():
        for word in words:
            for m in re.finditer(rf"\b{word}\b", lowered):
                before = lowered[: m.start()].split()[-3:]
                if any(neg in before for neg in _NEGATION):
                    avoided = period  # type: ignore[assignment]
                else:
                    preferred = period  # type: ignore[assignment]
    return preferred, avoided


def _extract_title(text: str) -> str | None:
    """strip request boilerplate, duration and date words; what remains is the activity name."""
    cleaned = _DURATION_RE.sub(" ", text)
    cleaned = _DATE_HINT_RE.sub(" ", cleaned)
    cleaned = re.sub(
        r"\b(xếp|xep|sắp xếp|sap xep|giúp|giup|cho|mình|minh|tôi|toi|vào|vao|không|khong|buổi|buoi|"
        r"sáng|sang|chiều|chieu|tối|toi|nhé|nhe|nha|please|schedule|for|me)\b",
        " ",
        cleaned,
        flags=re.IGNORECASE,
    )
    cleaned = re.sub(r"[^\w\s]", " ", cleaned)
    cleaned = re.sub(r"\s+", " ", cleaned).strip()
    return cleaned[:255] or None


def rule_based_understand(text: str) -> NluResult:
    """
    input: text (str) sanitized prompt.
    core logic: regex extraction of duration, time-of-day, date hint, priority and a coarse intent.
    output: NluResult(source="rules"); every field is None when not literally present in the text.
    """
    lowered = text.lower()
    duration = extract_duration_minutes(text)
    preferred, avoided = _time_of_day_flags(text)
    date_match = _DATE_HINT_RE.search(text)

    if re.search(r"\b(xóa|xoa|hủy|huy|delete|cancel)\b", lowered):
        intent: Intent = "delete_event"
    elif re.search(r"\b(dời|doi|đổi|chuyển|chuyen|move|reschedule)\b", lowered):
        intent = "reschedule"
    elif duration is not None or re.search(r"\b(xếp|xep|sắp xếp|sap xep)\b", lowered):
        intent = "schedule_task"
    elif re.search(r"\b(lịch|lich|có gì|co gi|khi nào|khi nao|mấy giờ|schedule|what)\b", lowered):
        intent = "query_schedule"
    else:
        intent = "unknown"

    priority = 1 if any(re.search(rf"\b{w}\b", lowered) for w in _HIGH_PRIORITY) else None
    return NluResult(
        intent=intent,
        title=_extract_title(text) if intent == "schedule_task" else None,
        duration_minutes=duration,
        priority=priority,
        preferred_time_of_day=preferred,
        avoid_time_of_day=avoided,
        date_hint=date_match.group(0) if date_match and intent != "unknown" else None,
        fallback_mode=False,
        source="rules",
    )

def _extract_json_object(raw: str) -> dict:
    """pull the first {...} block out of a model reply and parse it (raises ValueError / JSONDecodeError)."""
    start, end = raw.find("{"), raw.rfind("}")
    if start == -1 or end <= start:
        raise ValueError("no json object in model output")
    parsed = json.loads(raw[start : end + 1])
    if not isinstance(parsed, dict):
        raise ValueError("json root is not an object")
    return parsed


def log_ai_error(
    db: Session | None,
    user_id: int | None,
    prompt: str,
    raw_output: str | None,
    error_type: AiErrorType,
    retry_count: int,
) -> None:
    """persist one AiErrorLog row (flush only; the request owner commits). No-op when db is None."""
    if db is None:
        return
    db.add(
        AiErrorLog(
            user_id=user_id,
            prompt_text=prompt,
            raw_output=(raw_output or "")[:4000] or None,
            error_type=error_type,
            retry_count=retry_count,
            triggered_fallback=True,
        )
    )
    db.flush()


def _finalize(result: NluResult, rules: NluResult, prompt: str, now: datetime, norm_profanity: bool) -> NluResult:
    """shared post-processing: literal-text guards, deterministic date window, clarification, profanity flag."""
    if rules.duration_minutes is not None:
        result.duration_minutes = rules.duration_minutes  # literal text wins over the model
    elif not _DURATION_RE.search(prompt):
        result.duration_minutes = None  # model invented a duration
    # temporal hallucination guard: only a date phrase that literally occurs in the prompt is accepted
    hint = rules.date_hint
    if hint is None and result.date_hint and result.date_hint.lower() in prompt.lower():
        hint = result.date_hint
    result.date_hint = hint
    if result.intent in {"schedule_task", "query_schedule", "reschedule"}:
        window = resolve_date_hint(hint, now)
        result.window_start, result.window_end = window.start, window.end
        result.date_assumed = window.assumed
        if window.clarification_question:
            result.needs_clarification, result.clarification_question = True, window.clarification_question
    if result.intent == "schedule_task" and result.duration_minutes is None:
        result.needs_clarification = True
        result.clarification_question = result.clarification_question or "Bạn muốn dành bao lâu cho hoạt động này?"
    if result.priority is None:
        result.priority = rules.priority
    result.profanity_found = norm_profanity
    return result


def understand_prompt(
    prompt: str,
    reference_now: datetime,
    *,
    db: Session | None = None,
    user_id: int | None = None,
    client: httpx.Client | None = None,
) -> NluResult:
    """
    input: prompt (str) raw user message, reference_now (aware datetime), optional db/user_id for error logs.
    core logic: sanitize + normalize (P2-6) -> RAG context (P2-2) -> few-shot llm call through the FIFO broker
                (P2-10) -> strict json/pydantic validation with one retry (P2-1) -> deterministic date window
                and clarification (P2-3). Any failure (TIMEOUT, INFRA_CRASH, JSON_PARSE/SCHEMA_VALIDATION,
                SEMANTIC_FAILURE) is logged to ai_error_logs + the jsonl taxonomy file (P2-12) and the
                rule-based result is returned with fallback_mode=True.
    output: NluResult. Never raises for AI failures, so the mobile UI can always open the manual picker.
    """
    if reference_now.tzinfo is None:
        raise ValueError("reference_now must be timezone-aware")
    started = time.monotonic()
    normalized = normalize_text(sanitize_prompt(prompt))
    safe_prompt = normalized.text
    if not safe_prompt:
        return NluResult(intent="unknown", fallback_mode=True, source="rules", profanity_found=normalized.profanity_found)

    base = rule_based_understand(safe_prompt)
    rules = _finalize(base.model_copy(), base, safe_prompt, reference_now, normalized.profanity_found)
    context: list[str] = []
    if user_id is not None:
        try:
            context = [h.text for h in rag.retrieve(user_id, safe_prompt, k=2)]
        except Exception:
            context = []  # a corrupt/missing index must never block the request
    now_vn = reference_now.astimezone(VN_TZ)
    system = f"{SYSTEM_PROMPT} Now: {now_vn:%Y-%m-%d %A %H:%M} (+07:00)."
    if context:
        system += " Known schedule: " + " | ".join(c.replace("\n", "; ")[:200] for c in context)
    messages = [{"role": "system", "content": system}, *FEW_SHOT, {"role": "user", "content": safe_prompt}]
    payload = {"model": MODEL_NAME, "messages": messages, "max_tokens": 200, "temperature": 0.0}

    http = client or httpx.Client(timeout=LLM_TIMEOUT_SECONDS)
    raw_output: str | None = None
    retries = 0
    category = "SEMANTIC_FAILURE"
    reason = "unresolvable intent"
    try:
        assert_private_endpoint(VLLM_ENDPOINT)
        while True:
            try:
                response = broker.run(
                    lambda: http.post(VLLM_ENDPOINT, json=payload, timeout=LLM_TIMEOUT_SECONDS), user_id=user_id
                )
                response.raise_for_status()
                raw_output = response.json()["choices"][0]["message"]["content"]
                result = NluResult.model_validate(_extract_json_object(raw_output))
            except (ValueError, ValidationError, KeyError, IndexError) as exc:
                category = "SCHEMA_VALIDATION" if isinstance(exc, ValidationError) else "JSON_PARSE"
                reason = f"{type(exc).__name__}: {str(exc)[:200]}"
                if retries < MAX_SCHEMA_RETRIES:
                    retries += 1  # reject malformed output and retry once
                    continue
                break
            if result.intent == "unknown" and rules.intent == "unknown":
                category, reason = "SEMANTIC_FAILURE", "model and rules could not identify an intent"
                break
            result.source, result.fallback_mode = "llm", False
            record_latency_ms((time.monotonic() - started) * 1000)
            return _finalize(result, rules, safe_prompt, reference_now, normalized.profanity_found)
    except (httpx.TimeoutException, QueueWaitTimeout) as exc:
        category, reason = "TIMEOUT", f"{type(exc).__name__}: {exc}"
    except (httpx.TransportError, httpx.HTTPStatusError, ValueError) as exc:
        category, reason = "INFRA_CRASH", f"{type(exc).__name__}: {str(exc)[:200]}"
    finally:
        if client is None:
            http.close()

    record = log_ai_failure(category, safe_prompt, reason, retrieved_context=context, raw_output=raw_output,
                            user_id=user_id, retry_count=retries)
    log_ai_error(db, user_id, safe_prompt, raw_output, AiErrorType(record["db_error_type"]), retries)
    record_latency_ms((time.monotonic() - started) * 1000)
    rules.fallback_mode = True
    return rules

# owner: Database & AI (P1-2 data privacy isolation: payload sanitization, outbound allowlist, local encryption)
import base64
import hashlib
import ipaddress
import os
import re
import unicodedata
from urllib.parse import urlparse

from cryptography.fernet import Fernet, InvalidToken

MAX_PROMPT_CHARS = 500  # keeps system+user prompt inside the 2048-token vllm context window

_CONTROL_CHARS_RE = re.compile(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]")
_EMAIL_RE = re.compile(r"[\w.+-]+@[\w-]+(?:\.[\w-]+)+")
_PHONE_RE = re.compile(r"(?<!\d)(?:\+?84|0)(?:[\s.-]?\d){8,10}(?!\d)")
_URL_RE = re.compile(r"https?://\S+", re.IGNORECASE)
_PRIVATE_HOST_SUFFIXES = (".local", ".internal", ".lan")
_CGNAT = ipaddress.ip_network("100.64.0.0/10")
_DOCKER_SERVICE_HOSTS = {"llama_engine", "local_llama_inference", "localhost"}


def sanitize_prompt(text: str, max_chars: int = MAX_PROMPT_CHARS) -> str:
    """
    input: text (str) raw user prompt and max_chars (int) hard length cap.
    core logic: NFC-normalize, strip control chars, redact emails/phones/urls, collapse whitespace, truncate.
    output: str safe to embed in an llm prompt or persist to ai_error_logs.
    typical flow: "mail a@b.com 0901234567" -> "mail [email] [phone]".
    """
    cleaned = unicodedata.normalize("NFC", text)
    cleaned = _CONTROL_CHARS_RE.sub(" ", cleaned)
    cleaned = _EMAIL_RE.sub("[email]", cleaned)
    cleaned = _URL_RE.sub("[url]", cleaned)
    cleaned = _PHONE_RE.sub("[phone]", cleaned)
    cleaned = re.sub(r"\s+", " ", cleaned).strip()
    return cleaned[:max_chars]


def is_private_endpoint(url: str) -> bool:
    """
    input: url (str) of the llm / internal service endpoint.
    core logic: accept only loopback, RFC1918/link-local ips, docker service names or *.local hosts.
    output: bool, True when the host is not a public internet address.
    typical flow: http://127.0.0.1:8000 -> True; https://api.openai.com -> False.
    """
    host = (urlparse(url).hostname or "").lower()
    if not host:
        return False
    if host in _DOCKER_SERVICE_HOSTS or host.endswith(_PRIVATE_HOST_SUFFIXES):
        return True
    allowed = [x.strip().lower() for x in os.getenv("LLM_ALLOWED_HOSTS", "").split(",") if x.strip()]
    if host in allowed:
        return True  # explicit opt-in for the team's own AI host (e.g. a Radmin/Hamachi 26.x.x.x address)
    try:
        addr = ipaddress.ip_address(host)
    except ValueError:
        return False  # unknown public-looking dns name
    if addr in _CGNAT:
        return True  # tailscale range
    for rule in allowed:
        if "/" in rule and addr in ipaddress.ip_network(rule, strict=False):
            return True  # allowlisted CIDR, e.g. 26.0.0.0/8
    return addr.is_private or addr.is_loopback or addr.is_link_local


def assert_private_endpoint(url: str) -> str:
    """
    input: url (str) outbound target for personal schedule data.
    core logic: raise instead of letting schedule data leave the private network.
    output: the same url when private; raises ValueError otherwise.
    """
    if not is_private_endpoint(url):
        raise ValueError(f"refusing to send schedule data to non-private endpoint: {url}")
    return url


def _fernet() -> Fernet:
    """derive a Fernet key from DATA_ENCRYPTION_KEY (falls back to a dev-only key)."""
    secret = os.getenv("DATA_ENCRYPTION_KEY", "dev-encryption-key-change-in-production")
    key = base64.urlsafe_b64encode(hashlib.sha256(secret.encode("utf-8")).digest())
    return Fernet(key)


def encrypt_text(plain: str) -> str:
    """
    input: plain (str) sensitive text such as a stored prompt.
    output: str urlsafe Fernet token (AES-128-CBC + HMAC) safe for a TEXT column.
    """
    return _fernet().encrypt(plain.encode("utf-8")).decode("ascii")


def decrypt_text(token: str) -> str | None:
    """
    input: token (str) produced by encrypt_text.
    output: original str, or None when the token is corrupted / key mismatched.
    """
    try:
        return _fernet().decrypt(token.encode("ascii")).decode("utf-8")
    except (InvalidToken, ValueError):
        return None

# owner: Database & AI (P2-2 RAG pipeline: file text -> chunks -> embeddings -> per-user encrypted vector index)
import json
import math
import os
import re
import threading
import unicodedata
import zlib
from dataclasses import dataclass
from pathlib import Path

from services.ai_logging import log_ai_failure
from services.privacy import decrypt_text, encrypt_text

RAG_DIR = Path(os.getenv("RAG_DATA_DIR", str(Path(__file__).resolve().parent.parent / "data" / "rag")))
EMBED_DIM = 512
CHUNK_CHARS = 400
CHUNK_OVERLAP = 80
MIN_SCORE = 0.12  # below this the retrieval is treated as a miss and logged for prompt tuning


def fold(text: str) -> str:
    """lowercase + strip Vietnamese diacritics so 'đồ án' and 'do an' embed identically."""
    nfd = unicodedata.normalize("NFD", text.lower().replace("đ", "d"))
    return "".join(c for c in nfd if unicodedata.category(c) != "Mn")


def embed(text: str, dim: int = EMBED_DIM) -> list[float]:
    """
    deterministic offline embedding (feature hashing of word unigrams/bigrams + char trigrams, L2-normalised).
    no network and no model download, so personal schedule text never leaves the host (P1-2). swap this one
    function for a sentence-transformer if a GPU budget is available; the index format stays the same.
    """
    words = re.findall(r"\w+", fold(text))
    feats = list(words) + [f"{a}_{b}" for a, b in zip(words, words[1:])]
    for w in words:
        padded = f"^{w}$"
        feats += [padded[i : i + 3] for i in range(len(padded) - 2)]
    vec = [0.0] * dim
    for f in feats:
        h = zlib.crc32(f.encode("utf-8"))
        vec[h % dim] += 1.0 if (h >> 16) & 1 else -1.0
    norm = math.sqrt(sum(x * x for x in vec)) or 1.0
    return [x / norm for x in vec]


def chunk_text(text: str, size: int = CHUNK_CHARS, overlap: int = CHUNK_OVERLAP) -> list[str]:
    """
    split extracted timetable text into chunks: whole lines are kept together (a timetable row is never cut),
    lines are packed up to `size` chars, and the last `overlap` chars are repeated into the next chunk.
    """
    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]
    chunks: list[str] = []
    current = ""
    for line in lines:
        line = line[:size]
        if current and len(current) + len(line) + 1 > size:
            chunks.append(current)
            current = current[-overlap:].split("\n", 1)[-1] if overlap else ""
        current = f"{current}\n{line}".strip("\n") if current else line
    if current:
        chunks.append(current)
    return chunks


@dataclass
class Hit:
    text: str
    score: float
    source: str


class VectorIndex:
    """per-owner vector store persisted as one Fernet-encrypted JSON file (local encryption, P1-2)."""

    _lock = threading.Lock()

    def __init__(self, owner_id: int, directory: Path | None = None) -> None:
        self.owner_id = owner_id
        self.path = (directory or RAG_DIR) / f"owner_{int(owner_id)}.idx"
        self._items: list[dict] = self._load()

    def _load(self) -> list[dict]:
        if not self.path.is_file():
            return []
        plain = decrypt_text(self.path.read_text(encoding="ascii"))
        return json.loads(plain) if plain else []

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(encrypt_text(json.dumps(self._items)), encoding="ascii")

    def __len__(self) -> int:
        return len(self._items)

    def replace_source(self, source: str, chunks: list[str]) -> int:
        """re-import safe: drop all chunks previously indexed from `source`, then add the new ones."""
        with self._lock:
            self._items = [i for i in self._items if i["source"] != source]
            self._items += [{"source": source, "text": c, "vec": embed(c)} for c in chunks]
            self._save()
        return len(chunks)

    def clear(self) -> None:
        with self._lock:
            self._items = []
            self.path.unlink(missing_ok=True)

    def search(self, query: str, k: int = 3) -> list[Hit]:
        q = embed(query)
        scored = [Hit(i["text"], sum(a * b for a, b in zip(q, i["vec"])), i["source"]) for i in self._items]
        return sorted(scored, key=lambda h: h.score, reverse=True)[:k]


def ingest_file(owner_id: int, file_bytes: bytes, filename: str, directory: Path | None = None) -> int:
    """
    input: owner_id, raw bytes of a .pdf/.png/.jpg/.csv/.md/.txt timetable, original filename.
    core logic: parser.extract_raw_text_from_file (pdf text layer / OCR for images) -> chunk -> embed -> index.
    output: number of chunks indexed. Extraction failure or empty text is logged as RAG_EXTRACTION_FAILED and
            raises ValueError so the API can tell the user instead of silently indexing nothing.
    """
    from services.parser import extract_raw_text_from_file  # lazy: pulls optional pdf/ocr deps

    try:
        text = extract_raw_text_from_file(file_bytes, filename)
    except Exception as exc:  # parser raises on unsupported / OCR-less environments
        log_ai_failure("RAG_EXTRACTION_FAILED", f"file:{filename}", f"{type(exc).__name__}: {exc}", user_id=owner_id)
        raise ValueError(f"cannot extract text from {filename}: {exc}") from exc
    chunks = chunk_text(text)
    if not chunks:
        log_ai_failure("RAG_EXTRACTION_FAILED", f"file:{filename}", "extracted text was empty", user_id=owner_id)
        raise ValueError(f"no readable text found in {filename}")
    return VectorIndex(owner_id, directory).replace_source(filename, chunks)


def retrieve(owner_id: int, query: str, k: int = 3, directory: Path | None = None) -> list[Hit]:
    """top-k chunks for `query`; logs RAG_EMPTY_RETRIEVAL / RAG_LOW_SCORE so prompts can be tuned later."""
    index = VectorIndex(owner_id, directory)
    if len(index) == 0:
        return []  # user has not indexed anything; not a failure
    hits = [h for h in index.search(query, k) if h.score >= MIN_SCORE]
    if not hits:
        best = index.search(query, 1)
        log_ai_failure("RAG_LOW_SCORE" if best else "RAG_EMPTY_RETRIEVAL", query,
                       f"best score {best[0].score:.3f} < {MIN_SCORE}" if best else "empty index", user_id=owner_id,
                       retrieved_context=[b.text for b in best])
    return hits

# owner: Database & AI (P2-3 temporal ambiguity resolution: deterministic defaults + clarifying questions)
import re
import unicodedata
from dataclasses import dataclass
from datetime import date, datetime, timedelta

from database import VN_TZ

_WEEKDAYS = {"hai": 0, "2": 0, "ba": 1, "3": 1, "tư": 2, "4": 2, "năm": 3, "5": 3,
             "sáu": 4, "6": 4, "bảy": 5, "7": 5}
_VAGUE = ("sớm", "lúc nào đó", "sắp tới", "dạo này", "khi nào rảnh", "một lúc nào", "soon", "sometime", "whenever")


@dataclass
class ResolvedWindow:
    start: date
    end: date  # inclusive
    assumed: bool = False  # True when a default was applied instead of an explicit user statement
    clarification_question: str | None = None
    note: str | None = None


def _week_monday(d: date) -> date:
    return d - timedelta(days=d.weekday())


def resolve_date_hint(hint: str | None, reference_now: datetime) -> ResolvedWindow:
    """
    input: hint (str|None) literal date words from the prompt, reference_now (aware datetime).
    core logic: map relative Vietnamese/English expressions to an inclusive VN-local date window using only
                reference_now (no model involvement => zero temporal hallucination). Vague phrases produce a
                clarifying question plus a safe default (next 7 days). Missing hint => default window, assumed.
    output: ResolvedWindow.
    """
    if reference_now.tzinfo is None:
        raise ValueError("reference_now must be timezone-aware")
    today = reference_now.astimezone(VN_TZ).date()
    default = ResolvedWindow(today, today + timedelta(days=6), assumed=True,
                             note="no date given; searching the next 7 days")
    if not hint or not hint.strip():
        return default
    h = unicodedata.normalize("NFC", hint).lower().strip()

    if any(v in h for v in _VAGUE):
        default.clarification_question = "Bạn muốn xếp vào ngày nào? (ví dụ: ngày mai, thứ sáu, tuần sau)"
        default.note = "vague date; defaulted to next 7 days"
        return default

    m = re.search(r"(\d{1,2})\s*[/-]\s*(\d{1,2})(?:\s*[/-]\s*(\d{2,4}))?", h)
    if m:
        day, month = int(m.group(1)), int(m.group(2))
        year = int(m.group(3)) if m.group(3) else today.year
        if year < 100:
            year += 2000
        try:
            d = date(year, month, day)
        except ValueError:
            default.clarification_question = f"Ngày {m.group(0)} không hợp lệ, bạn muốn ngày nào?"
            return default
        if not m.group(3) and d < today:
            d = date(year + 1, month, day)
        return ResolvedWindow(d, d)

    if "hôm nay" in h or h == "today":
        return ResolvedWindow(today, today)
    if "ngày mai" in h or h == "tomorrow":
        return ResolvedWindow(today + timedelta(days=1), today + timedelta(days=1))
    if "ngày kia" in h:
        return ResolvedWindow(today + timedelta(days=2), today + timedelta(days=2))

    next_week = "tuần sau" in h or "next week" in h
    wd = re.search(r"thứ\s*(hai|ba|tư|năm|sáu|bảy|[2-7])", h)
    sunday = "chủ nhật" in h and not wd

    if next_week and not wd and not sunday:
        mon = _week_monday(today) + timedelta(days=7)
        return ResolvedWindow(mon, mon + timedelta(days=6))
    if "tuần này" in h or "this week" in h:
        return ResolvedWindow(today, _week_monday(today) + timedelta(days=6))
    if "cuối tuần" in h:
        sat = _week_monday(today) + timedelta(days=5)
        return ResolvedWindow(max(sat, today), sat + timedelta(days=1))
    if "tháng sau" in h:
        first = (today.replace(day=1) + timedelta(days=32)).replace(day=1)
        return ResolvedWindow(first, (first + timedelta(days=32)).replace(day=1) - timedelta(days=1))
    if "cuối tháng" in h:
        last = (today.replace(day=1) + timedelta(days=32)).replace(day=1) - timedelta(days=1)
        return ResolvedWindow(max(today, last - timedelta(days=6)), last)

    if wd or sunday:
        target = 6 if sunday else _WEEKDAYS[wd.group(1)]
        if next_week:
            d = _week_monday(today) + timedelta(days=7 + target)
            return ResolvedWindow(d, d)
        d = today + timedelta(days=(target - today.weekday()) % 7)
        window = ResolvedWindow(d, d)
        if d == today:
            window.assumed = True
            window.note = "weekday is today; assumed today (say 'tuần sau' for next week)"
        return window

    default.clarification_question = f'Mình chưa hiểu "{hint}". Bạn muốn xếp vào ngày nào?'
    return default

@dataclass
class NormalizedText:
    text: str
    original: str
    profanity_found: bool = False
    changes: list[str] = field(default_factory=list)


def _word_sub(text: str, mapping: dict[str, str], changes: list[str], tag: str) -> str:
    """replace whole words/phrases (case-insensitive), longest keys first, recording each change."""
    for key in sorted(mapping, key=len, reverse=True):
        value = mapping[key]
        if key == value:
            continue
        pattern = re.compile(rf"(?<![\w]){re.escape(key)}(?![\w])", re.IGNORECASE)
        new_text, n = pattern.subn(value, text)
        if n:
            changes.append(f"{tag}:{key}->{value}")
            text = new_text
    return text


def normalize_text(raw: str) -> NormalizedText:
    """
    input: raw (str) user prompt, possibly without diacritics, teencode, emoji or profanity.
    core logic: NFC -> drop emoji/unsupported symbols -> squeeze repeated letters -> remove profanity ->
                restore diacritics for known scheduling phrases -> teencode dictionary -> collapse spaces.
    output: NormalizedText(text, original, profanity_found, changes). Never raises.
    typical flow: "xep lich 3 tieng lam do an ko buoi toi nhe" -> "xếp lịch 3 tiếng làm đồ án không buổi tối nhé".
    """
    changes: list[str] = []
    text = unicodedata.normalize("NFC", raw or "")
    stripped = _ALLOWED.sub(" ", _EMOJI_SYMBOLS.sub(" ", text))
    if stripped != text:
        changes.append("special_chars_removed")
    text = _REPEAT.sub(r"\1\1", stripped)

    profanity = False
    kept: list[str] = []
    for word in text.split():
        if word.lower().strip(".,!?") in PROFANITY:
            profanity = True
            continue
        kept.append(word)
    if profanity:
        changes.append("profanity_removed")
    text = " ".join(kept)

    text = _word_sub(text, RESTORE_PHRASES, changes, "diacritics")
    text = _word_sub(text, TEENCODE, changes, "teencode")
    text = re.sub(r"\s+", " ", text).strip()
    return NormalizedText(text=text, original=raw, profanity_found=profanity, changes=changes)

# owner: Database & AI (P2-10 FIFO request broker serializing llm inference to avoid OOM / context switching)
import threading
import time
import uuid
from collections import deque
from collections.abc import Callable
from datetime import datetime
from typing import TypeVar

from database import UTC_TZ
from models import InferenceQueueLog, QueueStatus

T = TypeVar("T")

QUEUE_MAX_WAIT_SECONDS = 1.0  # + 3.0s llm timeout keeps worst-case end-to-end latency at 4s


class QueueWaitTimeout(Exception):
    """raised when a request waited longer than max_wait before reaching the model."""


class InferenceBroker:
    """
    strict first-in-first-out gate: exactly one inference runs at a time, the rest wait in arrival order.
    a ticket deque + condition variable (not a plain Lock) guarantees FIFO fairness.
    """

    def __init__(self, recorder: Callable[[dict], None] | None = None) -> None:
        self._cond = threading.Condition()
        self._waiting: deque[str] = deque()
        self._running: str | None = None
        self.recorder = recorder
        self.completed = 0

    @property
    def depth(self) -> int:
        """requests currently waiting or running."""
        with self._cond:
            return len(self._waiting) + (1 if self._running else 0)

    def run(self, fn: Callable[[], T], *, user_id: int | None = None, max_wait: float = QUEUE_MAX_WAIT_SECONDS) -> T:
        """
        input: fn (callable doing the model call), user_id for logging, max_wait seconds allowed in line.
        core logic: enqueue ticket -> wait until it is at the head and the model is idle -> run fn -> release.
        output: fn's return value; raises QueueWaitTimeout (status TIMEOUT) or re-raises fn's error (FAILED).
        """
        ticket = uuid.uuid4().hex
        enq = datetime.now(UTC_TZ)
        deadline = time.monotonic() + max_wait
        with self._cond:
            self._waiting.append(ticket)
            while not (self._running is None and self._waiting[0] == ticket):
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    self._waiting.remove(ticket)
                    self._cond.notify_all()
                    self._record(ticket, user_id, enq, None, datetime.now(UTC_TZ), QueueStatus.TIMEOUT)
                    raise QueueWaitTimeout(f"waited more than {max_wait}s for the model")
                self._cond.wait(timeout=remaining)
            self._waiting.popleft()
            self._running = ticket
        started = datetime.now(UTC_TZ)
        status = QueueStatus.COMPLETED
        try:
            return fn()
        except BaseException:
            status = QueueStatus.FAILED
            raise
        finally:
            with self._cond:
                self._running = None
                self.completed += 1
                self._cond.notify_all()
            self._record(ticket, user_id, enq, started, datetime.now(UTC_TZ), status)

    def _record(self, ticket: str, user_id: int | None, enq: datetime, start: datetime | None,
                end: datetime, status: QueueStatus) -> None:
        if self.recorder is None:
            return
        try:
            self.recorder({"request_id": ticket, "user_id": user_id, "enqueued_at": enq,
                           "started_at": start, "completed_at": end, "status": status})
        except Exception:
            pass  # telemetry must never break inference


def db_recorder(entry: dict) -> None:
    """persist one InferenceQueueLog row in its own short session (timestamps stored as naive utc)."""
    from database import SessionLocal  # local import: keeps unit tests free of a db connection

    with SessionLocal() as session:
        row = {k: (v.replace(tzinfo=None) if isinstance(v, datetime) else v) for k, v in entry.items()}
        session.add(InferenceQueueLog(**row))
        session.commit()

# owner: Database & AI (P2-11 IT telemetry: latency, RAM, VRAM, queue depth -> telemetry_metric_logs)
import os

import subprocess
import threading
import time
from collections import deque
from collections.abc import Callable

from services.inference_queue import broker

try:  # optional dependency; sampler degrades gracefully without it
    import psutil
except ImportError:  # pragma: no cover
    psutil = None

_latencies: deque[float] = deque(maxlen=500)  # ms, end-to-end nlu latency


def record_latency_ms(ms: float) -> None:
    """call after each AI request; the sampler reports p50/p95 of the recent window."""
    _latencies.append(ms)


def _percentile(values: list[float], pct: float) -> float:
    ordered = sorted(values)
    return ordered[min(len(ordered) - 1, int(round(pct / 100 * (len(ordered) - 1))))]


def _gpu_memory_mb() -> tuple[float, float] | None:
    """(used, total) VRAM in MB via nvidia-smi, or None when no NVIDIA GPU / tool."""
    if shutil.which("nvidia-smi") is None:
        return None
    try:
        out = subprocess.run(
            ["nvidia-smi", "--query-gpu=memory.used,memory.total", "--format=csv,noheader,nounits"],
            capture_output=True, text=True, timeout=2, check=True,
        ).stdout.strip().splitlines()[0]
        used, total = (float(x) for x in out.split(","))
        return used, total
    except (subprocess.SubprocessError, OSError, ValueError, IndexError):
        return None


def collect_sample() -> list[tuple[str, float, dict | None]]:
    """
    output: list of (metric_name, value, metadata) for this instant: queue_depth, ram_used_mb, ram_percent,
            vram_used_mb, vram_percent, nlu_latency_p50_ms, nlu_latency_p95_ms (metrics are skipped when unavailable).
    """
    sample: list[tuple[str, float, dict | None]] = [("queue_depth", float(broker.depth), None)]
    if psutil is not None:
        mem = psutil.virtual_memory()
        sample += [("ram_used_mb", mem.used / 1048576, None), ("ram_percent", float(mem.percent), None)]
    gpu = _gpu_memory_mb()
    if gpu:
        sample += [("vram_used_mb", gpu[0], {"total_mb": gpu[1]}), ("vram_percent", 100 * gpu[0] / gpu[1], None)]
    if _latencies:
        data = list(_latencies)
        sample += [("nlu_latency_p50_ms", _percentile(data, 50), {"n": len(data)}),
                   ("nlu_latency_p95_ms", _percentile(data, 95), {"n": len(data)})]
    return sample


def write_sample_to_db() -> int:
    """persist one sample to telemetry_metric_logs; returns the number of rows written."""
    from database import SessionLocal
    from models import TelemetryMetricLog

    rows = collect_sample()
    with SessionLocal() as session:
        session.add_all(TelemetryMetricLog(metric_name=n, metric_value=v, metadata_json=m) for n, v, m in rows)
        session.commit()
    return len(rows)


class TelemetryCollector:
    """background daemon thread sampling every interval_seconds; start()/stop() are idempotent."""

    def __init__(self, interval_seconds: float, writer: Callable[[], int] = write_sample_to_db) -> None:
        self.interval = interval_seconds
        self._writer = writer
        self._stop = threading.Event()
        self._thread: threading.Thread | None = None

    def start(self) -> None:
        if self.interval <= 0 or (self._thread and self._thread.is_alive()):
            return
        self._stop.clear()
        self._thread = threading.Thread(target=self._loop, name="telemetry", daemon=True)
        self._thread.start()

    def stop(self) -> None:
        self._stop.set()
        if self._thread:
            self._thread.join(timeout=2)

    def _loop(self) -> None:
        while not self._stop.is_set():
            try:
                self._writer()
            except Exception:
                pass  # db down must not kill the sampler
            self._stop.wait(self.interval)


collector = TelemetryCollector(float(os.getenv("TELEMETRY_INTERVAL_SECONDS", "0")))  # 0 = disabled

LOG_PATH = Path(os.getenv("AI_ERROR_LOG_PATH", str(Path(__file__).resolve().parent.parent / "logs" / "ai_errors.jsonl")))

# fine-grained taxonomy -> coarse DB enum (models.py is locked, so the exact category lives in the file log)
CATEGORY_TO_DB: dict[str, AiErrorType] = {
    "TIMEOUT": AiErrorType.TIMEOUT,
    "INFRA_CRASH": AiErrorType.INFRA_CRASH,
    "JSON_PARSE": AiErrorType.SCHEMA_VALIDATION,
    "SCHEMA_VALIDATION": AiErrorType.SCHEMA_VALIDATION,
    "TEMPORAL_HALLUCINATION": AiErrorType.SEMANTIC_FAILURE,
    "SEMANTIC_FAILURE": AiErrorType.SEMANTIC_FAILURE,
    "RAG_EMPTY_RETRIEVAL": AiErrorType.SEMANTIC_FAILURE,
    "RAG_LOW_SCORE": AiErrorType.SEMANTIC_FAILURE,
    "RAG_EXTRACTION_FAILED": AiErrorType.SEMANTIC_FAILURE,
}


def log_ai_failure(
    category: str,
    prompt: str,
    reason: str,
    *,
    retrieved_context: list[str] | None = None,
    raw_output: str | None = None,
    user_id: int | None = None,
    retry_count: int = 0,
) -> dict:
    """
    input: category (key of CATEGORY_TO_DB), the exact prompt, failure reason, retrieved RAG chunks, raw model output.
    core logic: append one JSON line (utc timestamp) to LOG_PATH under a lock; failures to write never propagate.
    output: the record dict that was written (also useful for tests).
    """
    if category not in CATEGORY_TO_DB:
        raise ValueError(f"unknown ai error category: {category}")
    record = {
        "ts": datetime.now(UTC_TZ).isoformat(),
        "category": category,
        "db_error_type": CATEGORY_TO_DB[category].value,
        "user_id": user_id,
        "prompt": sanitize_prompt(prompt, max_chars=2000),
        "retrieved_context": [sanitize_prompt(c, max_chars=600) for c in (retrieved_context or [])],
        "raw_output": (raw_output or "")[:4000] or None,
        "reason": reason[:500],
        "retry_count": retry_count,
    }
    try:
        with _lock:
            LOG_PATH.parent.mkdir(parents=True, exist_ok=True)
            with LOG_PATH.open("a", encoding="utf-8") as fh:
                fh.write(json.dumps(record, ensure_ascii=False) + "\n")
    except OSError:
        pass  # logging must never break a request
    return record

@router.post("/index-file")
async def index_file(
    file: UploadFile = File(...),
    current_user: User = Depends(get_current_user),
) -> dict[str, int | str]:
    """
    input: authenticated upload of a timetable pdf / image / text file.
    core logic: extract text (pdf layer or OCR) -> chunk -> embed -> replace this file's chunks in the
                caller's own encrypted vector index (re-upload of the same filename replaces, never duplicates).
    output: number of chunks indexed; 422 when nothing readable was found.
    """
    data = await file.read(MAX_UPLOAD_BYTES + 1)
    if len(data) > MAX_UPLOAD_BYTES:
        raise HTTPException(413, "file too large (max 10 MB)")
    try:
        count = rag.ingest_file(current_user.id, data, file.filename or "upload")
    except ValueError as exc:
        raise HTTPException(422, str(exc)) from exc
    return {"filename": file.filename or "upload", "chunks_indexed": count}

# owner: Database & AI (P2-13 reminder scheduler audit trail; notifications themselves fire locally on the phone)
import json
import os
from datetime import datetime, timedelta
from pathlib import Path

from fastapi import APIRouter, Depends, HTTPException, status
from pydantic import BaseModel, Field
from sqlalchemy import select
from sqlalchemy.orm import Session

from database import UTC_TZ, active_events_stmt, from_db_to_vn, get_db, to_db_utc
from models import Event, Reminder, User
from services.security import get_current_user

router = APIRouter(prefix="/api/v1/reminders", tags=["reminders"])

AUDIT_LOG_PATH = Path(os.getenv("REMINDER_AUDIT_LOG_PATH", str(Path(__file__).resolve().parent.parent / "logs" / "reminder_audit.jsonl")))


class ReminderUpsert(BaseModel):
    offset_minutes: int = Field(default=30, ge=0, le=7 * 24 * 60)


class DeviceReport(BaseModel):
    at: datetime | None = None  # device-side timestamp (aware); defaults to server now


class ReminderOut(BaseModel):
    reminder_id: int
    event_id: int
    offset_minutes: int
    fire_at: str  # ISO 8601 +07:00; the phone schedules its local alarm here
    scheduled_on_device_at: str | None
    delivered_at: str | None


def _audit(action: str, reminder: Reminder, owner_id: int) -> None:
    """append one line to the audit file; a disk error must never fail the request."""
    record = {"ts": datetime.now(UTC_TZ).isoformat(), "action": action, "reminder_id": reminder.id,
              "event_id": reminder.event_id, "owner_id": owner_id, "offset_minutes": reminder.offset_minutes}
    try:
        AUDIT_LOG_PATH.parent.mkdir(parents=True, exist_ok=True)
        with AUDIT_LOG_PATH.open("a", encoding="utf-8") as fh:
            fh.write(json.dumps(record) + "\n")
    except OSError:
        pass


def _get_event(db: Session, owner_id: int, event_id: int) -> Event:
    event = db.scalar(active_events_stmt(owner_id).where(Event.id == event_id))  # owner + soft-delete filtered
    if event is None:
        raise HTTPException(status.HTTP_404_NOT_FOUND, "event not found")
    return event


def _get_reminder(db: Session, owner_id: int, event_id: int) -> Reminder:
    _get_event(db, owner_id, event_id)
    reminder = db.scalar(select(Reminder).where(Reminder.event_id == event_id, Reminder.owner_id == owner_id))
    if reminder is None:
        raise HTTPException(status.HTTP_404_NOT_FOUND, "reminder not found")
    return reminder


def _out(reminder: Reminder, event: Event) -> ReminderOut:
    fire = from_db_to_vn(event.start_time) - timedelta(minutes=reminder.offset_minutes)
    return ReminderOut(
        reminder_id=reminder.id, event_id=reminder.event_id, offset_minutes=reminder.offset_minutes,
        fire_at=fire.isoformat(),
        scheduled_on_device_at=from_db_to_vn(reminder.scheduled_on_device_at).isoformat() if reminder.scheduled_on_device_at else None,
        delivered_at=from_db_to_vn(reminder.delivered_at).isoformat() if reminder.delivered_at else None,
    )


@router.put("/{event_id}", response_model=ReminderOut)
def upsert_reminder(event_id: int, body: ReminderUpsert, db: Session = Depends(get_db),
                    current_user: User = Depends(get_current_user)) -> ReminderOut:
    """create or change the single reminder of an event; resets device timestamps so the phone must re-report."""
    event = _get_event(db, current_user.id, event_id)
    reminder = db.scalar(select(Reminder).where(Reminder.event_id == event_id, Reminder.owner_id == current_user.id))
    action = "updated"
    if reminder is None:
        reminder, action = Reminder(owner_id=current_user.id, event_id=event_id, offset_minutes=body.offset_minutes), "created"
        db.add(reminder)
    else:
        reminder.offset_minutes = body.offset_minutes
        reminder.scheduled_on_device_at = reminder.delivered_at = None
    db.commit()
    db.refresh(reminder)
    _audit(action, reminder, current_user.id)
    return _out(reminder, event)


def _report(field: str, action: str, event_id: int, body: DeviceReport, db: Session, user: User) -> ReminderOut:
    reminder = _get_reminder(db, user.id, event_id)
    setattr(reminder, field, to_db_utc(body.at or datetime.now(UTC_TZ)))
    db.commit()
    db.refresh(reminder)
    _audit(action, reminder, user.id)
    return _out(reminder, _get_event(db, user.id, event_id))


@router.post("/{event_id}/scheduled", response_model=ReminderOut)
def report_scheduled(event_id: int, body: DeviceReport, db: Session = Depends(get_db),
                     current_user: User = Depends(get_current_user)) -> ReminderOut:
    """the phone confirms its local alarm was registered (AlarmManager / WorkManager)."""
    return _report("scheduled_on_device_at", "scheduled", event_id, body, db, current_user)


@router.post("/{event_id}/delivered", response_model=ReminderOut)
def report_delivered(event_id: int, body: DeviceReport, db: Session = Depends(get_db),
                     current_user: User = Depends(get_current_user)) -> ReminderOut:
    """the phone confirms the notification was actually shown."""
    return _report("delivered_at", "delivered", event_id, body, db, current_user)


@router.get("/audit", response_model=list[ReminderOut])
def audit_trail(db: Session = Depends(get_db), current_user: User = Depends(get_current_user)) -> list[ReminderOut]:
    """the caller's reminders with scheduled/delivered timestamps; delivered_at None past fire_at = missed."""
    rows = db.execute(
        select(Reminder, Event).join(Event, Event.id == Reminder.event_id)
        .where(Reminder.owner_id == current_user.id, Event.owner_id == current_user.id, Event.deleted_at.is_(None))
        .order_by(Event.start_time)
    ).all()
    return [_out(r, e) for r, e in rows]

## 1. Hướng dẫn Sử dụng Fixture Kiểm thử (`conftest.py`)

File `conftest.py` chứa các công cụ dùng chung do `pytest` tự động nạp. Các thành viên khi viết test cho module của mình **không cần khởi tạo kết nối database thủ công**, mà chỉ cần truyền tên fixture vào tham số của hàm test.

### 1.1. Ba Fixture có sẵn và Khi nào cần dùng

| Tên Fixture | Kiểu dữ liệu | Phạm vi (`scope`) | Mục đích & Khi nào sử dụng |
| :--- | :--- | :--- | :--- |
| `mysql_engine` | `sqlalchemy.Engine` | `session` (toàn đợt chạy) | Tự động kết nối cổng `3307`, kiểm tra `SELECT 1` và tạo sẵn 11 bảng. Thường chỉ dùng nội bộ trong `conftest.py`, thành viên ít khi cần gọi trực tiếp. |
| `db_session` | `sqlalchemy.orm.Session` | `function` (từng bài test) | **Dùng khi cần truy vấn, thêm, sửa, xóa dữ liệu MySQL trực tiếp bằng SQLAlchemy ORM.** Mỗi bài test được bọc trong một giao dịch độc lập, tự động rollback khi test xong nên không làm bẩn DBeaver. |
| `client` | `fastapi.testclient.TestClient` | `function` (từng bài test) | **Dùng khi cần giả lập gửi request HTTP (`client.post`, `client.get`) vào các API Router.** `client` này đã được gắn sẵn với `db_session` cô lập ở trên. |

---

### 1.2. Hướng dẫn Cụ thể cho Thành viên Viết Test

#### Trường hợp 1: Hàm test logic thuần (không đụng tới Database hay API)
Không cần truyền bất kỳ fixture nào vào tham số:
```python
def test_password_hashing():
    # kiểm tra hàm băm mật khẩu hoặc thuật toán thuần python
    digest = hash_password("MyPassword123")
    assert verify_password("MyPassword123", digest) is True
```

#### Trường hợp 2: Hàm test cần thao tác trực tiếp với bảng trong MySQL
Khai báo tham số `db_session: Session`:
```python
from sqlalchemy.orm import Session
from models import User

def test_create_user_directly(db_session: Session):
    # tạo và lưu trực tiếp qua ORM
    new_user = User(display_name="Trần Văn A")
    db_session.add(new_user)
    db_session.commit()
    db_session.refresh(new_user)

    # kiểm tra dữ liệu đã vào MySQL
    assert new_user.id is not None
    # khi hàm test này kết thúc, giao dịch tự động rollback, dòng trên biến mất khỏi DBeaver
```

#### Trường hợp 3: Hàm test gọi qua đường API HTTP (FastAPI Endpoints)
Khai báo tham số `client: TestClient`:
```python
from fastapi.testclient import TestClient

def test_login_api(client: TestClient):
    # gửi request json qua client
    res = client.post("/api/v1/auth/login", json={"username": "test_user", "password": "123"})
    assert res.status_code in (200, 401)
```

#### Trường hợp 4: Hàm test vừa gọi API vừa cần kiểm tra bảng MySQL
Khai báo cả hai tham số `client: TestClient, db_session: Session`:
```python
from fastapi.testclient import TestClient
from sqlalchemy.orm import Session
from sqlalchemy import select
from models import User

def test_register_and_verify_database(client: TestClient, db_session: Session):
    # 1. Gọi API đăng ký
    res = client.post("/api/v1/auth/register", json={
        "username": "sinhvien_01",
        "password": "Password@123",
        "display_name": "Nguyễn Văn B"
    })
    assert res.status_code == 201
    user_id = res.json()["user_id"]

    # 2. Dùng db_session truy vấn trực tiếp vào MySQL để kiểm tra tính toàn vẹn
    saved_user = db_session.scalar(select(User).where(User.id == user_id))
    assert saved_user is not None
    assert saved_user.display_name == "Nguyễn Văn B"
```

### 1.3. Hai Lưu ý Quan trọng khi Chạy Test trên MySQL Thật

1. **Bộ đếm `AUTO_INCREMENT` vẫn tăng:**
   Trong cơ chế của MySQL InnoDB, số tự tăng (`id`) được cấp phát độc lập ở bộ nhớ bảng để tối ưu hiệu năng và không quay lùi khi rollback. Do đó, khi viết câu lệnh `assert`, **luôn so sánh theo biến `saved_user.id` động**, không được gán cứng giả định `assert saved_user.id == 1`.
2. **Cách giữ lại dữ liệu để mở DBeaver xem trực quan:**
   Mặc định sau khi test xong, dữ liệu tự hủy để giữ sạch database. Nếu bạn muốn giữ lại dữ liệu vừa test để bấm F5 trên DBeaver xem tận mắt các dòng được ghi thế nào:
   * Trên PowerShell, gõ: `$env:PERSIST_TEST_DATA="1"` trước khi chạy `pytest`.
   * Xem xong, xóa biến môi trường: `Remove-Item Env:PERSIST_TEST_DATA`.

### 2.1. File `test_auth.py` (Khôi phụ trách)
Kiểm thử trực tiếp trên MySQL 8.0 đối với cụm module: **Hạ tầng Cơ sở dữ liệu (`database.py`)** và **Hệ thống Xác thực (`services/security.py`, `routers/auth.py`)**.

**5 kịch bản kiểm chứng:**
1. **An toàn băm mật khẩu (`test_password_hashing_and_verification`):**
   * Cùng một mật khẩu băm 2 lần bằng `bcrypt` phải sinh ra 2 chuỗi digest khác nhau (nhờ muối ngẫu nhiên) và không bao giờ trùng chuỗi gốc.
   * Xác thực đúng mật khẩu trả về `True`, sai mật khẩu trả về `False`, và gặp chuỗi hash hỏng định dạng vẫn trả về `False` an toàn thay vì làm sập tiến trình (`ValueError`).
2. **Vòng đời JWT Token & Chốt chặn `get_current_user` (`test_jwt_lifecycle_and_current_user_guard`):**
   * Lưu bản ghi `User` tiếng Việt có dấu vào MySQL, ký JWT và xác nhận `get_current_user` truy xuất đúng bản ghi từ MySQL.
   * Từ chối bằng mã `HTTP 401` đối với cả 3 trường hợp: token hết hạn (`expires_delta < 0`), token bị sửa chữ ký, và token hợp lệ nhưng `user_id` không tồn tại trong bảng `users`.
3. **Tính nguyên tử 3 bảng khi Đăng ký & Đăng nhập trên MySQL (`test_register_login_and_mysql_three_table_atomicity`):**
   * Gọi `POST /api/v1/auth/register` (mã `201`) tạo đồng bộ cả 3 bảng `users`, `auth_identities` (`provider = PASSWORD`) và `user_preferences` (`sleep_start = 23:30`, `sleep_end = 06:30`, `slot_step_minutes = 15`), bảo toàn tiếng Việt `utf8mb4`.
   * Người dùng tự đặt `username` nguyên bản, hệ thống không tự ý gắn thêm bất kỳ tiền tố nào (`prefix`).
   * Đăng ký trùng `username` trả về `HTTP 409 Conflict`; đăng nhập đúng trả về `HTTP 200`, sai mật khẩu trả về `HTTP 401`.
4. **Đổi username bên trong app & Ràng buộc không trùng lặp (`test_update_username_and_uniqueness`):**
   * Gọi `PATCH /api/v1/auth/username` để đổi username sau khi đã đăng nhập.
   * Đổi sang tên chưa ai dùng thành công (mã `200`); đổi sang tên đã thuộc về tài khoản khác bị chặn ngay bằng mã `HTTP 409 Conflict`.
5. **Chuẩn hóa múi giờ UTC/+07:00, Chống IDOR & Ràng buộc `CHECK` của MySQL 8.0 (`test_mysql_utc_roundtrip_check_constraints_and_owner_isolation`):**
   * Sự kiện lúc `07:30 +07:00` được lưu xuống cột `DATETIME` của MySQL thành `00:30` (UTC) và đọc ngược lên khớp đúng `07:30 +07:00`.
   * Hàm `active_events_stmt` cô lập tuyệt đối dữ liệu giữa hai tài khoản (`Sinh Viên A` và `Sinh Viên B`).
   * Ghi thử `priority = 9` vào bảng `events` bị động cơ MySQL 8.0 chặn ngay nhờ ràng buộc `chk_event_priority`.

### 2.4. File `test_search_engine.py` (P.Nguyên phụ trách)

**Các nhóm kịch bản bắt buộc phải bao phủ:**
1. **Chuẩn hóa truy vấn ngôn ngữ tự nhiên & Thời gian tương đối (`optimize_search_query`):**

2. **Truy vấn cục bộ trên MySQL `utf8mb4` & Chống lộ dữ liệu chéo (`execute_local_search` & `verify_and_format_search_results`):**
   * Lưu dữ liệu thật vào bảng `events` qua `db_session`; xác nhận truy vấn tiếng Việt tìm đúng bản ghi thuộc `owner_id` hiện tại, bỏ qua các sự kiện đã xóa mềm (`deleted_at IS NOT NULL`) và không bao giờ trả về lịch của sinh viên khác.
   * Mọi mốc thời gian trả ra JSON mang chuẩn ISO 8601 ở múi giờ `+07:00`, cam kết 0% ảo giác do không dùng mô hình sinh tự do để trả lời trực tiếp dữ liệu lịch.
3. **Độ chính xác và Hiệu năng `< 200ms` của `find_free_slots`:**
   * Trừ đúng các khoảng bận (`CLASS`, `EXAM`, `EVENT`, `TASK`), trừ khung giờ ngủ (`UserPreference.sleep_start` tới `sleep_end`), trừ danh sách các khung giờ người dùng đã từ chối trong phiên xếp lịch (`excluded_slots`), và cộng đúng khoảng nghỉ chuyển tiếp (`min_break_minutes = 15`).
   * Không bị chặn bởi `DEADLINE` (thời lượng bằng 0, `blocks_availability = False`), làm tròn giờ bắt đầu lên lưới 15 phút (`:00`, `:15`, `:30`, `:45`).
   * **Đo kiểm hiệu năng:** Trên tập 300 sự kiện của cả học kỳ 15 tuần truy vấn từ MySQL, thời gian tính toán của `find_free_slots` bắt buộc nhỏ hơn `200ms`.

3. **Thiếu gói thư viện Python (`ModuleNotFoundError` hoặc thiếu `cryptography` khi nối MySQL 8.0):**
   * MySQL 8.0 sử dụng cơ chế xác thực `caching_sha2_password` nên `pymysql` bắt buộc cần gói `cryptography`. Chạy lệnh cài đặt đầy đủ:
     `python -m pip install fastapi uvicorn sqlalchemy pymysql cryptography bcrypt pyjwt python-multipart pytest httpx`

### Trường hợp 4: `DESELECTED` / `SKIPPED`

def test_register_login_and_mysql_three_table_atomicity(
    client: TestClient, db_session: Session
) -> None:
    """
    input: client (TestClient) and db_session (Session) connected to docker mysql 8.0.
    core logic: register user without prefixes, verify 3 mysql tables + 409 on duplicate + login 200/401.
    output: assertions proving atomic 3-table creation and strict duplicate rejection on mysql.

def test_update_username_and_uniqueness(
    client: TestClient, db_session: Session
) -> None:
    """
    input: client (TestClient) and db_session (Session) with two distinct registered users.
    core logic: change username via PATCH /username, reject collision with existing user, allow unique change.
    output: assertions proving username can be updated and uniqueness is enforced across all accounts.
    typical flow: user_a updates to unused name -> 200; user_a updates to user_b name -> 409 Conflict.
    """
    user_a_name = _unique_username("user_alpha")
    user_b_name = _unique_username("user_beta")

    # register user A
    res_a = client.post(
        "/api/v1/auth/register",
        json={"username": user_a_name, "password": "Password123!", "display_name": "User Alpha"},
    )

    headers_a = {"Authorization": f"Bearer {res_a.json()['access_token']}"}

    # register user B
    res_b = client.post(

        json={"username": user_b_name, "password": "Password123!", "display_name": "User Beta"},

        headers=headers_a,

    assert "already taken" in conflict_res.json()["detail"]

    identity = db_session.scalar(
        select(AuthIdentity).where(AuthIdentity.user_id == res_a.json()["user_id"])
    )
    assert identity is not None and identity.provider_user_id == updated_name


def test_mysql_utc_roundtrip_check_constraints_and_owner_isolation(
    db_session: Session,
) -> None:
    """
    input: db_session (Session) executing directly against mysql 8.0 innodb engine.
    core logic: verify +07:00 -> utc storage roundtrip, owner_id isolation, and mysql CHECK constraints.
    output: assertions confirming exact timezone conversion, anti-idor filtering, and error on bad priority.
    typical flow: create 2 users -> save +07:00 event as 00:30 utc -> query per owner -> test invalid priority=9.
    """
    user_a = User(display_name="Sinh Viên A")
    user_b = User(display_name="Sinh Viên B")
    db_session.add_all([user_a, user_b])
    db_session.commit()

    # 07:30 +07:00 on 2026-11-09 must be stored as 00:30 naive utc in mysql DATETIME
    vn_start = datetime(2026, 11, 9, 7, 30, tzinfo=VN_TZ)
    vn_end = datetime(2026, 11, 9, 9, 30, tzinfo=VN_TZ)
    ev_a = Event(
        owner_id=user_a.id,
        title="Cấu trúc dữ liệu và giải thuật",
        room="E201",
        kind=EventKind.CLASS,
        source=EventSource.IMPORTED,
        start_time=to_db_utc(vn_start),
        end_time=to_db_utc(vn_end),
        is_hard=True,
        priority=int(PriorityLevel.HIGH),
    )
    db_session.add(ev_a)
    db_session.commit()
    db_session.refresh(ev_a)

    assert ev_a.start_time == datetime(2026, 11, 9, 0, 30)  # stored as naive utc in mysql
    assert from_db_to_vn(ev_a.start_time) == vn_start  # restored to exact +07:00

    # anti-idor check: user_b must see zero events from user_a
    events_for_a = db_session.scalars(active_events_stmt(user_a.id)).all()
    events_for_b = db_session.scalars(active_events_stmt(user_b.id)).all()
    assert len(events_for_a) == 1
    assert len(events_for_b) == 0

    # mysql 8.0 native CHECK constraint chk_event_priority (priority BETWEEN 1 AND 3)
    nested = db_session.begin_nested()
    bad_event = Event(
        owner_id=user_a.id,
        title="Sự kiện sai độ ưu tiên",
        kind=EventKind.EVENT,
        source=EventSource.MANUAL,
        start_time=to_db_utc(vn_start),
        end_time=to_db_utc(vn_end),
        is_hard=False,
        priority=9,  # violates chk_event_priority on real mysql 8.0
    )
    db_session.add(bad_event)
    with pytest.raises((IntegrityError, OperationalError)):
        db_session.flush()
    nested.rollback()

# ---- P2-10 FIFO broker -----------------------------------------------------
def test_broker_is_fifo_and_serial():
    broker, order, active, peak = InferenceBroker(), [], [0], [0]
    gate = threading.Event()

    def job(i):
        def fn():
            active[0] += 1
            peak[0] = max(peak[0], active[0])
            if i == 0:
                gate.wait(2)
            order.append(i)
            active[0] -= 1
        return fn

    threads = []
    for i in range(5):
        t = threading.Thread(target=lambda i=i: broker.run(job(i), max_wait=5))
        t.start()
        threads.append(t)
        time.sleep(0.05)  # fix arrival order
    assert broker.depth == 5
    gate.set()
    for t in threads:
        t.join()
    assert order == [0, 1, 2, 3, 4] and peak[0] == 1 and broker.depth == 0


def test_broker_wait_timeout_and_records():
    rec = []
    broker = InferenceBroker(recorder=rec.append)
    gate = threading.Event()
    t = threading.Thread(target=lambda: broker.run(lambda: gate.wait(2)))
    t.start()
    time.sleep(0.05)
    with pytest.raises(QueueWaitTimeout):
        broker.run(lambda: None, max_wait=0.1)
    gate.set()
    t.join()
    assert [r["status"].value for r in rec] == ["TIMEOUT", "COMPLETED"]
    assert broker.depth == 0
